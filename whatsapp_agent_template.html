WhatsApp QR Code
Agent Blueprint

A complete, production-ready recipe to add unofficial WhatsApp Web QR login and message sending to any Node.js agent. Zero developer accounts, zero Meta approval — just scan and send.

📦 Package Source

Install from GitHub main branch, not npm. The npm release is frequently broken by WhatsApp Web layout updates.

🖥️ Browser Strategy

Skip Puppeteer's Chromium download. Use the local Google Chrome already installed on the host machine.

🔒 Session Persistence

Sessions are cached in .wwebjs_auth/. Users scan the QR code once — it stays linked across restarts.

📞 Phone Formatting

Indian numbers are auto-formatted from local (0XXXXXXXXXX) to international (91XXXXXXXXXX) before sending.

1

Installation Command

Run this in your project root. The environment variable prevents downloading a 120MB Chromium binary and avoids SSL certificate errors on Windows.

Terminal (PowerShell) Shell 
# Set env var to skip browser download, then install from GitHub
$env:PUPPETEER_SKIP_DOWNLOAD="true"
npm install github:pedroslopez/whatsapp-web.js qrcode
⚠️ Do NOT use npm install whatsapp-web.js — the stable npm release has broken code injections caused by recent WhatsApp Web internal updates. Always install from the GitHub main branch.

whatsapp-client.js — Full File

This is the complete background client manager. It finds Chrome, initializes the headless browser, generates QR codes, broadcasts connection events via SSE, and sends messages with auto-formatted phone numbers.

whatsapp-client.js JavaScript (ESM) 
import pkg from 'whatsapp-web.js';
const { Client, LocalAuth } = pkg;
import qrcode from 'qrcode';
import path from 'path';
import fs from 'fs';

let client = null;
let qrCodeDataUrl = null;
let connectionStatus = 'Disconnected';
let broadcastFn = null;

// ── Chrome Finder ──────────────────────────────────
function getChromeExecutablePath() {
  const commonPaths = [
    'C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe',
    'C:\\Program Files (x86)\\Google\\Chrome\\Application\\chrome.exe',
    path.join(process.env.LOCALAPPDATA || '', 'Google/Chrome/Application/chrome.exe'),
    path.join(process.env.USERPROFILE || '', 'AppData/Local/Google/Chrome/Application/chrome.exe')
  ];
  for (const p of commonPaths) {
    if (fs.existsSync(p)) { console.log('Found Chrome at:', p); return p; }
  }
  console.warn('Chrome not found in common paths.');
  return undefined;
}

// ── Status Getter ──────────────────────────────────
export function getWhatsappStatus() {
  return { status: connectionStatus, qr: qrCodeDataUrl };
}

// ── Init Client ───────────────────────────────────
export function initWhatsappClient(broadcastCallback) {
  if (client) return;
  broadcastFn = broadcastCallback;
  console.log('Initializing WhatsApp client...');

  client = new Client({
    authStrategy: new LocalAuth({
      dataPath: path.join(process.cwd(), '.wwebjs_auth')
    }),
    puppeteer: {
      headless: true,
      executablePath: getChromeExecutablePath(),
      args: ['--no-sandbox', '--disable-setuid-sandbox']
    }
  });

  // QR Code event
  client.on('qr', async (qr) => {
    connectionStatus = 'QR_Ready';
    qrCodeDataUrl = await qrcode.toDataURL(qr);
    if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'QR_Ready', qr: qrCodeDataUrl });
  });

  // Ready event
  client.on('ready', () => {
    connectionStatus = 'Connected';
    qrCodeDataUrl = null;
    console.log('WhatsApp client is ready!');
    if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Connected' });
  });

  // Auth failure event
  client.on('auth_failure', (msg) => {
    connectionStatus = 'Disconnected';
    console.error('WhatsApp auth failure:', msg);
    if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Disconnected' });
  });

  // Disconnected event
  client.on('disconnected', (reason) => {
    connectionStatus = 'Disconnected';
    console.log('WhatsApp disconnected:', reason);
    if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Disconnected' });
    client = null;
  });

  connectionStatus = 'Connecting';
  if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Connecting' });

  client.initialize().catch((err) => {
    connectionStatus = 'Disconnected';
    console.error('Error initializing WhatsApp client:', err);
    if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Disconnected' });
  });
}

// ── Disconnect ─────────────────────────────────────
export async function disconnectWhatsapp() {
  if (!client) return;
  await client.destroy();
  client = null;
  connectionStatus = 'Disconnected';
  qrCodeDataUrl = null;
  if (broadcastFn) broadcastFn({ type: 'wa-status', status: 'Disconnected' });
  initWhatsappClient(broadcastFn);
}

// ── Send Message ──────────────────────────────────
export async function sendWhatsappMessage(phone, messageText) {
  if (!client || connectionStatus !== 'Connected')
    throw new Error('WhatsApp is not linked. Scan QR first.');

  let cleanPhone = phone.replace(/\D/g, '');
  if (!cleanPhone) throw new Error('Invalid phone number.');

  // Auto-format Indian numbers
  if (cleanPhone.startsWith('0') && cleanPhone.length === 11)
    cleanPhone = '91' + cleanPhone.substring(1);
  else if (cleanPhone.length === 10)
    cleanPhone = '91' + cleanPhone;

  const chatId = `${cleanPhone}@c.us`;
  return await client.sendMessage(chatId, messageText);
}

server.js — Express Routes

Add these imports and routes to any Express server. The client initializes on startup, serves QR/status over REST, and broadcasts events via SSE.

server.js (add to your existing server) JavaScript 
// ── Import at top of server.js ─────────────────────
import {
  initWhatsappClient,
  getWhatsappStatus,
  disconnectWhatsapp
} from './whatsapp-client.js';

// ── API Routes ─────────────────────────────────────

// Get QR code & connection status
app.get('/api/whatsapp/status', (req, res) => {
  res.json(getWhatsappStatus());
});

// Disconnect / Logout
app.post('/api/whatsapp/disconnect', async (req, res) => {
  try {
    await disconnectWhatsapp();
    res.json({ success: true });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// ── Initialize on startup (before app.listen) ─────
initWhatsappClient((event) => broadcast(event));

app.js — Frontend Handlers

Add these functions to your frontend JavaScript. They poll the status on page load, listen for SSE broadcasts, and update the QR image / status badge dynamically.

app.js (add to your frontend script) JavaScript 
// ── Load status on page load ──────────────────────
async function loadWaStatus() {
  const res = await fetch('/api/whatsapp/status');
  const data = await res.json();
  updateWaUI(data);
}

// ── Update UI based on status ─────────────────────
function updateWaUI(data) {
  const text = document.getElementById('wa-status-text');
  const qrContainer = document.getElementById('wa-qr-container');
  const qrImg = document.getElementById('wa-qr-img');
  const loader = document.getElementById('wa-loader');
  const btnDisconnect = document.getElementById('btn-wa-disconnect');

  qrContainer.style.display = 'none';
  loader.style.display = 'none';
  btnDisconnect.classList.add('hidden');

  if (data.status === 'Connected') {
    text.innerText = 'Connected';
    btnDisconnect.classList.remove('hidden');
  } else if (data.status === 'QR_Ready' && data.qr) {
    text.innerText = 'Scan QR Code';
    qrImg.src = data.qr;
    qrContainer.style.display = 'flex';
  } else if (data.status === 'Connecting') {
    text.innerText = 'Connecting...';
    loader.style.display = 'block';
  } else {
    text.innerText = 'Disconnected';
  }
}

// ── SSE listener (add inside setupEventSource) ───
// case 'wa-status': updateWaUI(data); break;

// ── Disconnect handler ────────────────────────────
async function handleWaDisconnect() {
  if (!confirm('Log out of WhatsApp?')) return;
  await fetch('/api/whatsapp/disconnect', { method: 'POST' });
  loadWaStatus();
}

Critical Gotchas

🚫 Do NOT use --single-process or --no-zygote Puppeteer flags. Modern Chrome requires multi-process execution. These flags cause Error: Requesting main frame too early! on startup.
🚫 Do NOT call client.isRegisteredUser() before sending. This method uses an internal WhatsApp evaluation that is frequently broken by web updates, causing the cryptic t: t error. Send directly via client.sendMessage() instead.
⚠️ Deep-merge database defaults. When upgrading an existing database with new settings fields (like waMessageTemplate), the database loader must deep-merge defaults into the existing settings, otherwise new fields will be undefined.
✅ SSL bypass for Windows. Set process.env.NODE_TLS_REJECT_UNAUTHORIZED = '0' in server.js to prevent antivirus TLS inspection from breaking outgoing API calls.
